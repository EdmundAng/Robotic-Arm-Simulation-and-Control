                JOINT_LIMITS = [(0,180), (-75,75)] # relative to horizontal axis (0 deg)

                def trans_matrix(theta_deg, length):
                    """Homogeneous transform for one revolute joint + link (planar)."""
                    t = np.radians(theta_deg)
                    c, s = np.cos(t), np.sin(t)
                    return np.array([
                        [c, -s, length * c],
                        [s,  c, length * s],
                        [0,  0, 1]
                    ])

                def forward_kinematics(angles_deg):
                    """Return list of joint (x, y) positions, base included."""
