                # 2-DOF Planar Arm (homogenous matrix method)

                import numpy as np

                LINK_LENGTHS = [2,1.5] # define link lengths
                JOINT_LIMITS = [(0,180), (-75,75)] # relative to horizontal axis (0 deg)

                def trans_matrix(theta_deg, length):
                    """Homogeneous transform for one revolute joint + link (planar)."""
                    t = np.radians(theta_deg)
                    c, s = np.cos(t), np.sin(t)
                    return np.array([
                        [c, -s, length * c],
                        [s,  c, length * s],
                        [0,  0, 1]
                    ])

                def forward_kinematics(angles_deg):
                    """Return list of joint (x, y) positions, base included."""
                    T = np.eye(3)
                    pts = [(0.0, 0.0)]
                    for theta, L in zip(angles_deg, LINK_LENGTHS):
                        T = T @ trans_matrix(theta, L)     # chain transforms
                        pts.append((T[0, 2], T[1, 2]))     # position = translation column
                    return pts
